        let itemIndex = 1;

        function atualizarValorUnitario(index) {
            const selectProduto = document.querySelector(`[name="itens[${index}][produto_id]"]`);
            const valorUnitarioInput = document.querySelector(`[name="itens[${index}][valor_unitario]"]`);
            const precoVenda = selectProduto.options[selectProduto.selectedIndex].dataset.preco;
            
            valorUnitarioInput.value = parseFloat(precoVenda).toFixed(2);
            calcularTotalItem(index);
        }

        function calcularTotalItem(index) {
            const quantidadeInput = document.querySelector(`[name="itens[${index}][quantidade]"]`);
            const valorUnitarioInput = document.querySelector(`[name="itens[${index}][valor_unitario]"]`);
            const valorTotalInput = document.querySelector(`[name="itens[${index}][valor_total]"]`);

            const quantidade = parseFloat(quantidadeInput.value) || 0;
            const valorUnitario = parseFloat(valorUnitarioInput.value) || 0;
            const valorTotal = quantidade * valorUnitario;

            valorTotalInput.value = valorTotal.toFixed(2);
            calcularValorTotalCompra();
        }

        function calcularValorTotalCompra() {
            let valorTotalCompra = 0;
            const valoresTotais = document.querySelectorAll('.valor-total');

            valoresTotais.forEach(item => {
                valorTotalCompra += parseFloat(item.value) || 0;
            });

            document.getElementById('valor-total-compra').innerText = valorTotalCompra.toFixed(2);
            document.getElementById('valor_total_compra').value = valorTotalCompra.toFixed(2);
        }

        function adicionarItem() {
            const container = document.getElementById('itens-container');
            const newItem = document.createElement('div');
            newItem.classList.add('item', 'form-row');
            newItem.innerHTML = `
                <select name="itens[${itemIndex}][produto_id]" class="produto-select" onchange="atualizarValorUnitario(${itemIndex})" required>
                    <option value="" disabled selected>Selecione um Produto</option>
                    @foreach($produtos as $produto)
                        <option value="{{ $produto->id }}" data-preco="{{ $produto->preco_venda }}">{{ $produto->nome }}</option>
                    @endforeach
                </select>
                <input type="number" name="itens[${itemIndex}][quantidade]" class="quantidade" placeholder="Quantidade" oninput="calcularTotalItem(${itemIndex})" required>
                <input type="number" name="itens[${itemIndex}][valor_unitario]" class="valor-unitario" step="0.01" readonly>
                <input type="number" name="itens[${itemIndex}][valor_total]" class="valor-total" step="0.01" readonly>
                <button type="button" class="btn btn-danger" onclick="removerItem(this)">Remover</button>
            `;
            container.appendChild(newItem);
            itemIndex++;
        }

        function removerItem(button) {
            button.parentElement.remove();
            calcularValorTotalCompra();
        }

        function verificarEstoque(produtoId, quantidadeSolicitada, itemElement) {
            $.ajax({
                url: '/verificar-estoque',  // URL para verificar o estoque
                method: 'GET',
                data: { produto_id: produtoId },
                success: function(response) {
                    var quantidadeEstoque = response.quantidade_estoque;

                    // Exibe alerta se a quantidade for inferior a 5
                    if (quantidadeEstoque < 5) {
                        alert("Este Produto está com " + quantidadeEstoque + " em estoque.");
                    }

                    // Bloqueia a venda se a quantidade solicitada for maior que o estoque disponível
                    if (quantidadeSolicitada > quantidadeEstoque) {
                        alert("Quantidade solicitada maior que o disponível em estoque.");
                        $(itemElement).find('input[name="quantidades[]"]').val(quantidadeEstoque); // Ajusta a quantidade para o máximo disponível
                        $('#botao-confirmar').prop('disabled', true);
                    } else {
                        $('#botao-confirmar').prop('disabled', false);
                    }
                }

    public function store(Request $request)
{
    $request->validate([
        'data_coleta' => 'required|date|before_or_equal:today',
        'controle_coleta' => 'required',
        'nome' => 'required',
        'endereco' => 'required',
        'numero' => 'required',
        'bairro' => 'required',
        'cidade' => 'required',
        'produtos' => 'required|array',
        'produtos.*.produto_id' => 'required|exists:produtos,id',
        'produtos.*.quantidade' => 'required|integer|min:1',
        'produtos.*.valor_unitario' => 'required|numeric|min:0',
    ]);

    DB::beginTransaction();
    try {
        // Calcular o valor total da movimentação somando o total de todos os itens
        $valorTotalMovimentacao = array_sum(array_map(function ($item) {
            return $item['quantidade'] * $item['valor_unitario'];
        }, $request->produtos));

        // Criar a movimentação principal
        $movimentacao = Movimentacao::create([
            'data_coleta' => $request->data_coleta,
            'controle_coleta' => $request->controle_coleta,
            'nome' => $request->nome,
            'endereco' => $request->endereco,
            'numero' => $request->numero,
            'bairro' => $request->bairro,
            'cidade' => $request->cidade,
            'cliente_id' => $request->cliente_id,
            'observacao' => $request->observacao,
            'forma_pagamento_id' => $request->forma_pagamento,
            'prazo_id' => $request->prazo,
            'valor_total' => $valorTotalMovimentacao,
            'quantidade' => array_sum(array_column($request->produtos, 'quantidade')),
        ]);

        // Processar cada item da movimentação
        foreach ($request->produtos as $item) {
            $produto = Produto::find($item['produto_id']);
            if ($produto) {
                // Verificar o estoque do produto antes de salvar
                if ($produto->quantidade_estoque < $item['quantidade']) {
                    DB::rollBack();
                    return redirect()->back()->withErrors(
                        'Quantidade solicitada para o produto "' . $produto->nome . '" é maior que o estoque disponível.'
                    );
                }

                // Salvar o item da movimentação
                $movimentacao->itens()->create([
                    'produto_id' => $item['produto_id'],
                    'quantidade' => $item['quantidade'],
                    'valor_unitario' => $item['valor_unitario'],
                    'valor_total' => $item['quantidade'] * $item['valor_unitario'],
                ]);

                 // Verificar e criar a conta a receber se a forma de pagamento não for 'à vista'
    if ($request->forma_pagamento != 'avista') {
        $prazoSelecionado = Prazo::findOrFail($request->prazo);
        $prazo = (int) filter_var($prazoSelecionado->prazo, FILTER_SANITIZE_NUMBER_INT);
        $dataVencimento = Carbon::parse($request->input('data_coleta'))->addDays($prazo);

        // Verifica se a data de vencimento já está vencida
        $statusConta = $dataVencimento->isPast() ? 'atrasado' : 'pendente';

        // Log de cálculo do prazo e data de vencimento
        Log::info('Calculando prazo para conta a receber:', [
            'data_coleta' => $request->input('data_coleta'), 
            'prazo' => $prazo, 
            'data_vencimento' => $dataVencimento,
            'status' => $statusConta
        ]);

        ContasAReceber::create([
            'cliente_id' => $request->cliente_id,
            'descricao' => $request->input('descricao', 'Sem descrição'),
            'valor' => $request->valor_total,
            'data_venda' => $request->input('data_coleta'),
            'data_vencimento' => $dataVencimento,
            'status' => $statusConta,
            'forma_pagamento_id' => $request->forma_pagamento,
            'prazo' => $prazoSelecionado->prazo,
        ]);
    }

                // Atualizar o estoque do produto
                $produto->quantidade_estoque -= $item['quantidade'];
                $produto->save();
            }
        }

        DB::commit();
        return redirect()->route('movimentacao.index')->with('success', 'Movimentação salva com sucesso.');

    } catch (\Exception $e) {
        DB::rollBack();
        Log::error('Erro ao salvar itens da movimentação:', ['error' => $e->getMessage()]);
        return redirect()->back()->withErrors('Erro ao salvar a movimentação.');
    }
}
